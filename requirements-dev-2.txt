# 3. Implement causal masking on the test sentence "the cat sat on the mat",
# display the attention matrix before and after masking, and explain the differences.

import torch
import torch.nn.functional as F

sentence = "the cat sat on the mat"
words = sentence.split()
n = len(words)

# Example attention scores
scores = torch.tensor([
    [2.0, 1.0, 0.5, 1.5, 0.8, 0.3],
    [1.0, 2.0, 1.2, 0.5, 1.0, 0.7],
    [0.5, 1.2, 2.0, 1.5, 0.9, 1.1],
    [1.5, 0.5, 1.5, 2.0, 1.2, 0.8],
    [0.8, 1.0, 0.9, 1.2, 2.0, 1.5],
    [0.3, 0.7, 1.1, 0.8, 1.5, 2.0]
])

# Attention BEFORE masking
before = F.softmax(scores, dim=-1)

# Causal mask: block future tokens
mask = torch.tril(torch.ones(n, n))
masked_scores = scores.masked_fill(mask == 0, float("-inf"))

# Attention AFTER masking
after = F.softmax(masked_scores, dim=-1)

print("Words:", words)

print("\nAttention BEFORE masking:")
print(before)

print("\nAttention AFTER causal masking:")
print(after)

print("\nDifference:")
print("Before: every word can attend to every word.")
print("After: each word can attend only to itself and previous words.")
print("Future-word attention becomes 0.")

# 5. Load a news article dataset, get document representations using Word2Vec,
# GloVe and BERT, implement KNN-based retrieval using cosine similarity and
# Euclidean distance, and evaluate using Precision@5, Recall@5 and MRR.

!pip install gensim transformers sentence-transformers scikit-learn -q

import numpy as np
from sklearn.datasets import fetch_20newsgroups
from sklearn.metrics.pairwise import cosine_similarity, euclidean_distances
from gensim.downloader import load
from sentence_transformers import SentenceTransformer

# Load news dataset
data = fetch_20newsgroups(
    subset="test",
    remove=("headers", "footers", "quotes")
)

docs = data.data[:500]

# ---------------- WORD2VEC ----------------
w2v = load("word2vec-google-news-300")

def word2vec(text):
    words = text.lower().split()
    vectors = [w2v[w] for w in words if w in w2v]
    return np.mean(vectors, axis=0) if vectors else np.zeros(300)

W = np.array([word2vec(x) for x in docs])

# ---------------- GLOVE ----------------
glove = load("glove-wiki-gigaword-100")

def glove_vec(text):
    words = text.lower().split()
    vectors = [glove[w] for w in words if w in glove]
    return np.mean(vectors, axis=0) if vectors else np.zeros(100)

G = np.array([glove_vec(x) for x in docs])

# ---------------- BERT ----------------
bert = SentenceTransformer("all-MiniLM-L6-v2")
B = bert.encode(docs)

# ---------------- KNN RETRIEVAL ----------------
def retrieve(X, query, metric="cosine", k=5):

    if metric == "cosine":
        scores = cosine_similarity(X, query.reshape(1, -1)).ravel()
        return np.argsort(scores)[::-1][:k]

    distances = euclidean_distances(X, query.reshape(1, -1)).ravel()
    return np.argsort(distances)[:k]

# Use first document as query
query = 0

for name, X in [("Word2Vec", W), ("GloVe", G), ("BERT", B)]:

    print("\n", name)

    for metric in ["cosine", "euclidean"]:

        q = X[query]
        results = retrieve(X, q, metric)

        print(metric, "Top 5:", results)

        # Evaluation
        relevant = set(np.where(data.target == data.target[query])[0])
        retrieved = set(results)

        precision = len(retrieved & relevant) / 5
        recall = len(retrieved & relevant) / len(relevant)

        # Reciprocal Rank
        rr = 0
        for rank, i in enumerate(results, 1):
            if i in relevant:
                rr = 1 / rank
                break

        print("Precision@5:", precision)
        print("Recall@5:", recall)
        print("MRR:", rr)

# 7. Multi-Hop RAG using HotpotQA: load questions/supporting documents,
# SentenceTransformer + FAISS retrieval, Hop-1 → query → Hop-2 → LLM answer,
# compare Single-Hop and 2-Hop using Exact Match and F1.

!pip install -q datasets sentence-transformers faiss-cpu transformers

from datasets import load_dataset
from sentence_transformers import SentenceTransformer
from transformers import pipeline
import faiss, numpy as np

d = load_dataset("hotpotqa","distractor",split="validation[:20]")

print("Questions:",len(d))
print("First:",d[0]["question"])
print("Answer:",d[0]["answer"])
print("Type:",d[0]["type"])

docs = [t+": "+" ".join(s) for x in d for t,s in
        zip(x["context"]["title"],x["context"]["sentences"])]

m = SentenceTransformer("all-MiniLM-L6-v2")
E = m.encode(docs)
faiss_index = faiss.IndexFlatL2(E.shape[1])
faiss_index.add(np.array(E,dtype="float32"))

def getdoc(q):
    _,i = faiss_index.search(m.encode([q]).astype("float32"),1)
    return docs[i[0][0]]

llm = pipeline("text2text-generation",model="google/flan-t5-small")

def f1(a,b):
    a,b=set(a.lower().split()),set(b.lower().split())
    n=len(a&b)
    return 2*n/(len(a)+len(b)) if a and b else 0

em1=em2=f_1=f_2=0

for x in d:
    q,ans=x["question"],x["answer"]
    d1=getdoc(q)

    a1=llm(f"{d1}\nQ:{q}\nA:",max_new_tokens=30)[0]["generated_text"]

    d2=getdoc(q+" "+d1)
    a2=llm(f"{d1}\n{d2}\nQ:{q}\nA:",max_new_tokens=30)[0]["generated_text"]

    em1+=a1.lower().strip()==ans.lower().strip()
    em2+=a2.lower().strip()==ans.lower().strip()
    f_1+=f1(a1,ans)
    f_2+=f1(a2,ans)

print("\nSingle-Hop: EM =",em1/len(d),"F1 =",f_1/len(d))
print("2-Hop:      EM =",em2/len(d),"F1 =",f_2/len(d))

# 11. Design a coding agent using LangChain and LangGraph that accepts a
# natural-language query, decomposes it into subtasks, maintains short-term
# memory, conditionally searches again if information is insufficient,
# invokes calculation when required, and generates the final answer with
# the agent's execution trace.

!pip install -q langchain-groq langgraph langchain-community duckduckgo-search

import os
from langchain_groq import ChatGroq
from langchain_core.tools import tool
from langchain_community.tools import DuckDuckGoSearchRun
from langgraph.prebuilt import create_react_agent
from langgraph.checkpoint.memory import MemorySaver

os.environ["GROQ_API_KEY"] = "YOUR_GROQ_API_KEY"

llm = ChatGroq(
    model="openai/gpt-oss-20b",
    temperature=0
)

@tool
def calculator(x: str):
    """Calculate an expression."""
    try:
        return str(eval(x))
    except:
        return "Calculation error"

search = DuckDuckGoSearchRun()

@tool
def web_search(q: str):
    """Search the web. Search again if information is insufficient."""
    r = search.run(q)
    return r if len(r) > 50 else "INSUFFICIENT_INFORMATION"

tools = [calculator, web_search]
memory = MemorySaver()

prompt = """
You are a coding agent.
1. Break complex queries into subtasks.
2. Use calculator for calculations.
3. Use web_search for information.
4. If information is insufficient, search again with a better query.
5. Remember previous results during execution.
6. Combine results and give the final answer.
"""

agent = create_react_agent(llm, tools, prompt=prompt, checkpointer=memory)

q = input("Enter query: ")

r = agent.invoke(
    {"messages": [("user", q)]},
    {"configurable": {"thread_id": "lab11"}}
)

print("\nFINAL ANSWER:")
print(r["messages"][-1].content)

print("\nEXECUTION TRACE:")
for m in r["messages"]:
    if getattr(m, "tool_calls", None):
        for t in m.tool_calls:
            print("→", t["name"], t["args"])
    elif m.type == "tool":
        print("→ Result:", m.content[:100])

# 13. Load a medical image dataset and pretrained model, convert image to text,

import medmnist, networkx as nx, numpy as np
from PIL import Image
from transformers import BlipProcessor, BlipForConditionalGeneration, pipeline
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity

# Medical image dataset
d = medmnist.PneumoniaMNIST(split="test", download=True)
img = Image.fromarray(d.imgs[0]).convert("RGB")
label = d.labels[0][0]

# Image -> text
p = BlipProcessor.from_pretrained("Salesforce/blip-image-captioning-base")
m = BlipForConditionalGeneration.from_pretrained("Salesforce/blip-image-captioning-base")
x = p(img, return_tensors="pt")
caption = p.decode(m.generate(**x, max_new_tokens=30)[0], skip_special_tokens=True)
print("Image text:", caption)

# Document graph
docs = [
    caption,
    "Pneumonia is an infection of the lungs.",
    "Chest X-rays are used to examine the lungs.",
    "Medical images should be interpreted by experts."
]
G = nx.Graph()
G.add_nodes_from(range(len(docs)))

# Retrieve using image context + question
q = input("Question: ")
embed = SentenceTransformer("all-MiniLM-L6-v2")
E = embed.encode(docs)
query = embed.encode([caption + " " + q])
score = cosine_similarity(query, E)[0]
ids = np.argsort(score)[-2:][::-1]

context = " ".join(docs[i] for i in ids)

# Generate answer
llm = pipeline("text2text-generation", model="google/flan-t5-small")
answer = llm(f"Context: {context}\nQuestion: {q}\nAnswer:",
              max_new_tokens=50)[0]["generated_text"]

print("Answer:", answer)

# Evaluation
retrieval = len(ids) / 2
quality = cosine_similarity(embed.encode([answer]), embed.encode([context]))[0][0]
hallucination = 1 - quality

print("\n--- EVALUATION ---")
print("Retrieval Score:", retrieval)
print("Answer Quality:", round(quality, 3))
print("Hallucination Rate:", round(hallucination, 3))
